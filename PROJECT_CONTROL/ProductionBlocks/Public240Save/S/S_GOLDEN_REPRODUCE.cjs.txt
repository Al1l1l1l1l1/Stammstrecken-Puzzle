'use strict';
const fs=require('fs'),cp=require('child_process'),crypto=require('crypto'),path=require('path');
const root=process.argv[2],original=process.argv[3];
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const steps=process.argv.slice(4);
const results=[];
for(const step of steps){
  cp.execFileSync(process.execPath,[path.join(root,'GOLDEN_REFERENCE_NODE_'+step+'.cjs'),root],{stdio:['ignore','pipe','pipe'],maxBuffer:10*1024*1024});
  const name='GOLDENS_PRECODEC_'+step+'.json',actual=fs.readFileSync(path.join(root,'PROJECT_CONTROL/ProductionBlocks/Public240Save/S',name)),expected=fs.readFileSync(path.join(original,name));
  results.push({name,byteLength:actual.length,actualSha256:hash(actual),expectedSha256:hash(expected),exactBytesEqual:actual.equals(expected)});
}
console.log(JSON.stringify({runtime:process.version,root,role:'Only ignored reproduction copies written; original S evidence unchanged',results},null,2));
if(results.some(r=>!r.exactBytesEqual))process.exitCode=1;
