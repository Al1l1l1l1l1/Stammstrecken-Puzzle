function canonical(value) {
  if (value === null) return 'null';
  if (typeof value === 'string') {
    for (let i=0;i<value.length;i++) {
      const n=value.charCodeAt(i);
      if(n>=0xD800&&n<=0xDBFF) { const low=value.charCodeAt(++i); if(!(low>=0xDC00&&low<=0xDFFF))throw Error('Unpaired high surrogate'); }
      else if(n>=0xDC00&&n<=0xDFFF)throw Error('Unpaired low surrogate');
    }
    return JSON.stringify(value);
  }
  if (typeof value === 'number') { if(!Number.isSafeInteger(value))throw Error('Not a safe integer');return String(value); }
  if (typeof value === 'boolean') return value?'true':'false';
  if (Array.isArray(value)) return '['+value.map(canonical).join(',')+']';
  const keys=Object.keys(value).sort();
  return '{'+keys.map(k=>canonical(k)+':'+canonical(value[k])).join(',')+'}';
}

function checkpoint(cp,p,puzzle) {
  exact(cp,checkpointKeys,p);identity(cp.identity,p+'.identity');
  check(guid.test(cp.attemptId)&&cp.attemptId!=='00000000-0000-0000-0000-000000000000',p+'.attemptId','nonempty Guid D');
  check(['FIRST_RUN','PRACTICE','REVISION','ENDLESS'].includes(cp.mode)&&['Ready','Active','Suspended','SolvedPendingCommit'].includes(cp.phase),p,'closed session enums');
  check(nonnegative(cp.revision)&&nonnegative(cp.activeElapsedMs)&&nonnegative(cp.correctionCount)&&nonnegative(cp.hintCount),p,'safe nonnegative counters');
  check(cp.appliedCommandIds.every(x=>guid.test(x)&&x!=='00000000-0000-0000-0000-000000000000')&&new Set(cp.appliedCommandIds).size===cp.appliedCommandIds.length,p+'.appliedCommandIds','unique nonempty Guid D');
  check(cp.revision===cp.appliedCommandIds.length,p,'revision equals successful commands');
  check(cp.undo.length<=256&&cp.undo.length<=cp.revision,p+'.undo','bounded history');
  if(!puzzle)return;
  const w=puzzle.grid.width,h=puzzle.grid.height;
  check(cp.cells.length===w*h&&cp.previouslySet.length===w*h,p,'grid-sized cells/history');
  const validCell=/^(UNSET|MARK_EMPTY|MARK_OCCUPIED|TRACK_NS|TRACK_EW|TRACK_NE|TRACK_ES|TRACK_SW|TRACK_WN)$/;
  check(cp.cells.every(x=>validCell.test(x))&&cp.previouslySet.every(x=>typeof x==='boolean'),p,'typed cells/history');
  check(cp.rulesetVersion===puzzle.rulesetVersion&&cp.identity.puzzleId===puzzle.puzzleId&&cp.identity.sha256===hash(puzzle),p,'checkpoint puzzle binding');
  let cells=cp.cells.slice();
  for(let j=cp.undo.length-1;j>=0;j--) {
    const diff=cp.undo[j];exact(diff,'changes',p+'.undo['+j+']');let last=-1;
    check(diff.changes.length>0,p+'.undo['+j+']','nonempty diff');
    for(const c of diff.changes){exact(c,'x,y,before,after',p+'.change');const idx=c.y*w+c.x;check(nonnegative(c.x)&&nonnegative(c.y)&&c.x<w&&c.y<h&&idx>last,p+'.change','inbounds sorted unique');check(c.before!==c.after&&validCell.test(c.before)&&validCell.test(c.after),p+'.change','valid nonempty cell transition');check(cells[idx]===c.after,p+'.undo['+j+']','reverse replay current==after');cells[idx]=c.before;last=idx;}
  }
  check(cp.cells.every((c,i)=>c==='UNSET'||cp.previouslySet[i]),p,'sticky previouslySet covers cells');
  check(cp.cells.every(c=>c!=='MARK_EMPTY')||cp.usedEmptyMarker,p,'sticky empty marker');
  check(cp.cells.every(c=>c!=='MARK_OCCUPIED')||cp.usedOccupiedMarker,p,'sticky occupied marker');
  check(eq(cp.appliedCommandIds,[...cp.appliedCommandIds].sort()),p,'stored Guid comparison order');
  check(cp.phase==='SolvedPendingCommit'?cp.solvedEventId==='puzzle-solved:'+cp.attemptId:cp.solvedEventId===null,p,'phase/solved ID');
  check(cp.startedAtUtc===null||utc(cp.startedAtUtc),p+'.startedAtUtc','UTC or explicit null');check(cp.lastSavedAtUtc===null||utc(cp.lastSavedAtUtc),p+'.lastSavedAtUtc','UTC or explicit null');
}
function ledger(payload,p){
  const cp=payload.ledgerCheckpoint;
  exact(cp,'ledgerContractVersion,throughSequence,balancePatience,entryChainHashSha256,sourceSaveGeneration,journalDiagnosticTail',p+'.checkpoint');
  check(cp.ledgerContractVersion===1&&nonnegative(cp.throughSequence)&&nonnegative(cp.balancePatience)&&nonnegative(cp.sourceSaveGeneration)&&hex.test(cp.entryChainHashSha256),p,'Contract1 checkpoint');
  let seq=cp.throughSequence,chain=cp.entryChainHashSha256,balance=cp.balancePatience;
  for(const e of payload.economyJournal){exact(e,entryKeys+',entryChainHashSha256',p+'.entry');const projection=Object.fromEntries(entryKeys.split(',').map(k=>[k,e[k]]));check(e.sequence===++seq,p,'contiguous suffix');chain=sha(Buffer.concat([Buffer.from(chain,'hex'),Buffer.from(canonical(projection))]));check(chain===e.entryChainHashSha256,p,'raw32 previous + Ei bytes');balance+=e.amount;check(nonnegative(balance),p,'nonnegative safe balance');const fields=[e.catalogId,e.catalogRevision,e.catalogHashSha256];check(fields.every(v=>v===null)||fields.every(v=>v!==null),p,'catalog all null or all populated');}
  check(cp.journalDiagnosticTail.length<=64&&cp.journalDiagnosticTail.every(e=>e.sequence<=cp.throughSequence),p,'diagnostic tail bounds');
  return {sequence:seq,balance,chain};
}
